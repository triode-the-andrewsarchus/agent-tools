use std::sync::Arc;
use tokio::sync::mpsc::{Receiver, Sender};

pub struct Worker {
    id: usize,
    receiver: Receiver<Job>,
    results: Sender<JobResult>,
}

impl Worker {
    pub fn new(id: usize, rx: Receiver<Job>, tx: Sender<JobResult>) -> Self {
        Self {
            id,
            receiver: rx,
            results: tx,
        }
    }

    pub async fn run(mut self) {
        println!("Worker {} initialized", self.id);
        while let Some(job) = self.receiver.recv().await {
            let res = self.execute(job).await;
            if let Err(e) = self.results.send(res).await {
                eprintln!("Failed to send worker result: {}", e);
                break;
            }
        }
    }

    async fn execute(&self, job: Job) -> JobResult {
        // Execute background workload with deterministic timing
        let output = format!("Job {} processed by worker {}", job.id, self.id);
        JobResult { id: job.id, status: 0, output }
    }
}
